{{- $book := . -}} {{- $coverURL := "" -}} {{- with $book.Params.cover -}} {{- $coverURL = . -}} {{- else -}} {{- with $book.Resources.GetMatch "cover.*" -}} {{- $coverURL = .RelPermalink -}} {{- else -}} {{- $bookSlug := path.Base (strings.TrimSuffix "/" $book.RelPermalink) -}} {{- $root := $book.Site.Params.bookkit.coverRoot | default "/images/books" -}} {{- $root = strings.TrimSuffix "/" $root -}} {{- $sourceRoot := $book.Site.Params.bookkit.coverSourceRoot | default "static/images/books" -}} {{- $sourceRoot = strings.TrimSuffix "/" $sourceRoot -}} {{- $candidateURL := printf "%s/%s/cover.png" $root $bookSlug -}} {{- $candidateFile := printf "%s/%s/cover.png" $sourceRoot $bookSlug -}} {{- if fileExists $candidateFile -}} {{- $coverURL = $candidateURL -}} {{- end -}} {{- end -}} {{- end -}} {{- return $coverURL -}}
